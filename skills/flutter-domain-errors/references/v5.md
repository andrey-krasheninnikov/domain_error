# domain_error 5.x

Use `Result<T, E>` with success first and error second. `DomainResult<T>` is
`Result<T, DomainError>`; `FutureDomainResult<T>` is its Future alias. Construct
branches with `Result.error` / `Result.success`; use `ResultError` /
`ResultSuccess` patterns with `error` / `value` fields. Read a getter only
after selecting its matching branch; `domainError` reads a failed domain result.

Pass `mapToDomainError` and optional observers directly as named capture
arguments. Sync observers, including the reporter, return `Null`. An async
function or a direct `void` tear-off fails type checking. Wrap synchronous void
calls in a block ending with `return null;`.

Result provides map, andThen, mapError, recover and their Async variants.
The FutureResult extension provides unsuffixed composition on Future<Result>.
It is not the pre-5.x domain Future alias. Async callbacks are awaited;
exceptions and failed source futures propagate unchanged. Use capture explicitly
when those exceptions belong in an operation outcome.

## Runnable example

Read [shared rules](rules.md) before adapting this example. The assertions cover
identity, diagnostic separation, routing, failure isolation, and completion.

```dart
import 'dart:async';

import 'package:domain_error/domain_error.dart';

/// A closed error hierarchy owned by this example's account feature.
sealed class AccountError extends DomainError {
  const AccountError({
    super.message,
    super.cause,
    super.stackTrace,
  });
}

/// An unavailable account with a stable identifier and optional diagnostics.
final class AccountUnavailable extends AccountError {
  const AccountUnavailable({
    super.message,
    super.cause,
    super.stackTrace,
  });

  @override
  String get typeIdentifier {
    return 'account.unavailable';
  }
}

/// A business failure whose immutable field participates in equality.
final class InvalidAccount extends AccountError {
  final String field;

  const InvalidAccount(this.field);

  @override
  String get typeIdentifier {
    return 'account.invalid';
  }

  /// Includes [field] alongside the base error identity.
  @override
  List<Object?> get props {
    return [
      ...super.props,
      field,
    ];
  }
}

/// Run with assertions enabled to verify the public 5.x contract.
Future<void> main() async {
  _checkIdentity();
  _checkSuccessfulCapture();
  _checkDomainCapture();

  await _checkRawCapture();
  await _checkMapperFailure();

  await _checkFuturePayload();
  await _checkStreamPayload();

  await _checkComposition();
}

/// Diagnostic causes do not change equality; business fields do.
void _checkIdentity() {
  final original = AccountUnavailable(
    message: 'Offline',
    cause: Object(),
  );
  final sameBusinessError = AccountUnavailable(
    message: 'Offline',
    cause: Object(),
  );

  assert(original == sameBusinessError);
  assert(const InvalidAccount('email') != const InvalidAccount('name'));
}

/// Success skips error callbacks and folds through the success branch.
void _checkSuccessfulCapture() {
  final events = <String>[];

  final DomainResult<int> success = captureResultSync<int>(
    () {
      return 7;
    },
    mapToDomainError: (raw, trace) {
      return AccountUnavailable(
        cause: raw,
        stackTrace: trace,
      );
    },
    onRawError: (raw, trace) {
      events.add('unexpected');
      return null;
    },
  );

  assert(success.isSuccess && success.value == 7);

  final folded = success.fold(
    (error) {
      return -1;
    },
    (value) {
      return value * 2;
    },
  );

  assert(folded == 14);
  assert(events.isEmpty);
}

/// A domain error keeps its identity even when its observers and reporter fail.
void _checkDomainCapture() {
  final original = AccountUnavailable(
    message: 'Offline',
    cause: Object(),
  );
  final events = <String>[];

  final captured = captureResultSync<int>(
    () {
      throw original;
    },
    mapToDomainError: (raw, trace) {
      throw StateError('unexpected mapper');
    },
    onDomainError: (error, trace) {
      events.add(error.typeIdentifier);
      throw StateError('observer unavailable');
    },
    onObserverError: (error, trace) {
      events.add('reporter');
      throw StateError('reporter unavailable');
    },
  );

  assert(identical(captured.domainError, original));
  assert(events.join(',') == 'account.unavailable,reporter');
}

/// Raw observation, failure reporting, and mapping complete in that order.
///
/// The mapped error does not trigger a second domain-error notification.
Future<void> _checkRawCapture() async {
  final events = <String>[];
  final raw = StateError('transport detail');
  final mappedError = AccountUnavailable(cause: raw);

  final mapped = await captureResult<int>(
    () async {
      throw raw;
    },
    onRawError: (error, trace) async {
      await Future<void>.delayed(Duration.zero);
      assert(identical(error, raw));
      events.add('raw');
      throw StateError('observer unavailable');
    },
    onObserverError: (error, trace) async {
      await Future<void>.delayed(Duration.zero);
      events.add('reporter');
      throw StateError('reporter unavailable');
    },
    mapToDomainError: (error, trace) async {
      await Future<void>.delayed(Duration.zero);
      events.add('mapper');
      return mappedError;
    },
    onDomainError: (error, trace) {
      events.add('unexpected domain observer');
    },
  );

  assert(identical(mapped.domainError, mappedError));
  assert(events.join(',') == 'raw,reporter,mapper');
}

/// Mapper failures escape with their original error object and stack trace.
Future<void> _checkMapperFailure() async {
  final raw = StateError('transport detail');
  final mapperFailure = StateError('mapper failure');
  final mapperTrace = StackTrace.fromString('mapper-origin');
  var mapperEscaped = false;

  try {
    await captureResult<int>(
      () {
        throw raw;
      },
      mapToDomainError: (error, trace) {
        Error.throwWithStackTrace(
          mapperFailure,
          mapperTrace,
        );
      },
    );
  } on Object catch (error, trace) {
    mapperEscaped = true;
    assert(identical(error, mapperFailure));
    assert(identical(trace, mapperTrace));
  }

  assert(mapperEscaped);
}

/// Synchronous capture stores the Future payload without awaiting completion.
///
/// The caller attaches a handler before completing the Future with an error.
Future<void> _checkFuturePayload() async {
  final raw = StateError('transport detail');
  final completion = Completer<int>();

  final futurePayload = captureResultSync<Future<int>>(
    () {
      return completion.future;
    },
    mapToDomainError: (error, trace) {
      return AccountUnavailable(
        cause: error,
      );
    },
  );

  assert(identical(futurePayload.value, completion.future));

  final handled = futurePayload.value.catchError(
    (Object error) {
      assert(identical(error, raw));
      return -1;
    },
  );

  completion.completeError(raw);
  assert(await handled == -1);
}

/// Capture covers stream creation; the subscriber handles later stream errors.
Future<void> _checkStreamPayload() async {
  final raw = StateError('transport detail');

  final streamPayload = captureResultSync<Stream<int>>(
    () {
      return Stream<int>.error(raw);
    },
    mapToDomainError: (error, trace) {
      return AccountUnavailable(
        cause: error,
      );
    },
  );

  var lateStreamError = false;

  try {
    await for (final _ in streamPayload.value) {}
  } on Object catch (error) {
    lateStreamError = identical(error, raw);
  }

  assert(lateStreamError);
}

/// Composition transforms success and recovers error payloads.
///
/// A failed source Future still propagates outside recovery.
Future<void> _checkComposition() async {
  final composed = await Future<Result<int, String>>.value(
    const Result<int, String>.success(2),
  ).andThen(
    (value) async {
      return Result<int, String>.success(value + 1);
    },
  ).map(
    (value) {
      return value * 2;
    },
  );

  assert(composed.value == 6);

  final recovered = const Result<int, String>.error('offline').mapError(
    (error) {
      return error.toUpperCase();
    },
  ).recover(
    (error) {
      return Result<int, String>.success(error.length);
    },
  );

  assert(recovered.value == 7);

  final preserved = await const Result<int, String>.error('offline').mapAsync(
    (value) async {
      return value + 1;
    },
  );

  assert(preserved.error == 'offline');

  final raw = StateError('transport detail');
  var failedFutureEscaped = false;

  try {
    await Future<Result<int, String>>.error(raw).recover(
      (error) {
        return const Result<int, String>.success(0);
      },
    );
  } on Object catch (error) {
    failedFutureEscaped = identical(error, raw);
  }

  assert(failedFutureEscaped);
}
```
